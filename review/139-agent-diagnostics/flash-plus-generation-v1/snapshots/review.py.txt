from __future__ import annotations

import json
from pathlib import Path


ROOT = Path(__file__).resolve().parent


def load(name: str) -> dict:
    return json.loads((ROOT / name).read_text(encoding="utf-8"))


def save_once(name: str, value: object) -> None:
    path = ROOT / name
    encoded = json.dumps(value, ensure_ascii=False, indent=2) + "\n"
    if path.exists() and path.read_text(encoding="utf-8") != encoded:
        raise ValueError(f"FROZEN_FILE_CHANGED:{name}")
    if not path.exists():
        path.write_text(encoded, encoding="utf-8")


def claim_keys(article: str, repeat: int, candidate: int, numbers: list[int]) -> list[str]:
    return [f"{article}:r{repeat}:k{candidate}:c{number}" for number in numbers]


def add_keys(target: dict, article: str, repeat: int, candidate: int, numbers: list[int], value: object) -> None:
    for key in claim_keys(article, repeat, candidate, numbers):
        if key in target:
            raise ValueError(f"DUPLICATE_OVERRIDE:{key}")
        target[key] = value

def review_claims(template: dict) -> None:
    unsupported = source_failures()
    modality = modality_failures()
    atomicity = atomicity_failures()
    excluded = selection_exclusions()
    duplicate = duplicates()
    keys = {row["key"] for row in template["claim_ratings"]}
    overrides = set(unsupported) | set(modality) | set(atomicity) | set(excluded) | set(duplicate)
    if not overrides <= keys:
        raise ValueError(f"UNKNOWN_CLAIM_OVERRIDES:{sorted(overrides - keys)}")
    for row in template["claim_ratings"]:
        key = row["key"]
        row["review_status"] = "REVIEWED"
        if key in unsupported:
            error_types, reason = unsupported[key]
            row["source_support"] = {"verdict": "FAIL", "error_types": error_types, "reason": reason}
        else:
            row["source_support"] = {"verdict": "PASS", "error_types": [], "reason": ""}
        row["modality"] = {"verdict": "FAIL", "reason": modality[key]} if key in modality else {"verdict": "PASS", "reason": ""}
        row["atomicity"] = {"verdict": "FAIL", "reason": atomicity[key]} if key in atomicity else {"verdict": "PASS", "reason": ""}
        if key in duplicate:
            row["selection"] = {"verdict": "EXCLUDED", "category": "DUPLICATE", "reason": "앞선 Claim과 같은 의미를 반복한다."}
            row["duplicate_of_claim_no"] = duplicate[key]
        elif key in excluded:
            category, reason = excluded[key]
            row["selection"] = {"verdict": "EXCLUDED", "category": category, "reason": reason}
        else:
            row["selection"] = {"verdict": "IN_SCOPE", "category": "IN_SCOPE", "reason": ""}


def review_facts(template: dict) -> None:
    specs = fact_specs()
    rows = {row["key"]: row for row in template["fact_ratings"]}
    if set(rows) != set(specs):
        raise ValueError("FACT_REVIEW_SET_CHANGED")
    for key, row in rows.items():
        verdict, mapped, covering, errors, reason = specs[key]
        row.update(
            review_status="REVIEWED",
            verdict=verdict,
            error_types=errors,
            mapped_claim_nos=mapped,
            covering_claim_nos=covering,
            reason=reason,
        )


def validate_claim(row: dict) -> None:
    verdicts = {
        row["source_support"]["verdict"], row["modality"]["verdict"],
        row["atomicity"]["verdict"], row["selection"]["verdict"],
    }
    if row["review_status"] != "REVIEWED" or "PENDING" in verdicts:
        raise ValueError(f"UNREVIEWED_CLAIM:{row['key']}")
    if row["selection"]["category"] == "DUPLICATE" and row["duplicate_of_claim_no"] is None:
        raise ValueError(f"DUPLICATE_WITHOUT_TARGET:{row['key']}")


def claim_key(row: dict, claim_no: int) -> str:
    return f"{row['article_id']}:r{row['repeat']}:k{row['candidate']}:c{claim_no}"


def validate_covering_claim(fact: dict, claim_no: int, claim: dict) -> None:
    verdicts = (
        claim["source_support"]["verdict"], claim["modality"]["verdict"],
        claim["atomicity"]["verdict"], claim["selection"]["verdict"],
    )
    if verdicts != ("PASS", "PASS", "PASS", "IN_SCOPE"):
        raise ValueError(f"INVALID_COVERING_CLAIM:{fact['key']}:{claim_no}")


def validate_fact_state(row: dict) -> None:
    if row["review_status"] != "REVIEWED":
        raise ValueError(f"UNREVIEWED_FACT:{row['key']}")
    if row["verdict"] == "PENDING":
        raise ValueError(f"UNREVIEWED_FACT:{row['key']}")
    if row["verdict"] == "PASS":
        if row["error_types"] or not row["covering_claim_nos"]:
            raise ValueError(f"INVALID_PASS_FACT:{row['key']}")
        return
    if not row["error_types"] or row["covering_claim_nos"] or not row["reason"]:
        raise ValueError(f"INVALID_NONPASS_FACT:{row['key']}")


def validate_fact(row: dict, claims: dict[str, dict]) -> None:
    validate_fact_state(row)
    for claim_no in row["mapped_claim_nos"]:
        if claim_key(row, claim_no) not in claims:
            raise ValueError(f"UNKNOWN_MAPPED_CLAIM:{row['key']}:{claim_no}")
    for claim_no in row["covering_claim_nos"]:
        validate_covering_claim(row, claim_no, claims[claim_key(row, claim_no)])


def validate(review: dict) -> None:
    if review["conditions_revealed"]:
        raise ValueError("CONDITIONS_REVEALED_EARLY")
    if len(review["fact_ratings"]) != 96 or len(review["claim_ratings"]) != 501:
        raise ValueError("REVIEW_ROW_COUNT_CHANGED")
    claims = {row["key"]: row for row in review["claim_ratings"]}
    for row in claims.values():
        validate_claim(row)
    for row in review["fact_ratings"]:
        validate_fact(row, claims)


def main() -> None:
    review = load("review-template.json")
    review_claims(review)
    review_facts(review)
    validate(review)
    save_once("blind-review.json", review)
    print(json.dumps({"status": "REVIEW_FROZEN", "facts": 96, "claims": 501}, ensure_ascii=False))


if __name__ == "__main__":
    main()
