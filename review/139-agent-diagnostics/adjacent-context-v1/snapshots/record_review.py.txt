"""Freeze the condition-blinded manual review for the #139 adjacent-ID pilot."""

from __future__ import annotations

from collections import Counter
import json
from pathlib import Path


ROOT = Path(__file__).resolve().parent
OMITTED = "REQUIRED_MEANING_OMITTED"
CHANGED = "UNSUPPORTED_OR_CHANGED_MEANING"
EVIDENCE = "EVIDENCE_INSUFFICIENT"
MODALITY = "MODALITY_OR_ATTRIBUTION_MISMATCH"


def key(article: str, repeat: int, candidate: int, source: str) -> tuple[str, int, int, str]:
    return article, repeat, candidate, source


def main() -> None:
    review = json.loads((ROOT / "review-template.json").read_text(encoding="utf-8"))
    judgments: dict[tuple[str, int, int, str], tuple[str, list[str], str]] = {}

    for repeat in (1, 2):
        for candidate in (1, 2):
            judgments[key("N01", repeat, candidate, "s5")] = (
                "FAIL", [EVIDENCE], "곽노정 사장의 전체 이름·소속을 문장에 썼지만 이를 확인하는 s4를 연결하지 않았다."
            )
            judgments[key("N01", repeat, candidate, "s6")] = (
                "FAIL", [EVIDENCE], "‘이어’의 발언자를 곽노정 사장으로 확인하는 s4·s5를 연결하지 않았다."
            )
    judgments[key("N01", 1, 2, "s21")] = (
        "PARTIAL", [MODALITY], "문장은 추진 중인 목표를 보존했지만 modality를 FACT로 분류했다."
    )

    for repeat in (1, 2):
        for candidate in (1, 2):
            judgments[key("N02", repeat, candidate, "s14")] = (
                "FAIL", [MODALITY], "박성현 CEO의 인용 중 적어도 개발 완료 사실을 귀속 없는 회사 사실로 바꿨다."
            )
            judgments[key("N02", repeat, candidate, "s22")] = (
                "PARTIAL", [OMITTED], "삼성전자의 SRAM 기술 경쟁력 강화 방안 소개를 OMIT_ALLOWED로 제외했다."
            )
    for candidate in (1, 2):
        judgments[key("N02", 1, candidate, "s28")] = (
            "PARTIAL", [MODALITY], "개발 추진 문장은 보존했지만 해당 requirement의 modality를 FACT로 분류했다."
        )
    for repeat in (1, 2):
        judgments[key("N02", repeat, 1, "s31")] = (
            "PARTIAL", [OMITTED], "디자인 솔루션 협력으로 생태계 기반을 넓히는 사실을 OMIT_ALLOWED로 제외했다."
        )
        judgments[key("N02", repeat, 2, "s31")] = (
            "FAIL", [EVIDENCE] + ([MODALITY] if repeat == 2 else []),
            "생략된 주체를 삼성전자로 풀어 썼지만 이를 확인하는 s29를 연결하지 않았다."
            + (" 문장은 현재 진행을 유지했지만 modality도 PLAN_OR_TARGET으로 분류했다." if repeat == 2 else ""),
        )

    for repeat in (1, 2):
        for candidate in (1, 2):
            judgments[key("N03", repeat, candidate, "s23")] = (
                "FAIL", [EVIDENCE, MODALITY],
                "‘그’가 젠슨 황임을 확인하는 s22를 연결하지 않았고, 평가 발언을 FACT로 분류했다.",
            )
    for candidate in (1, 2):
        judgments[key("N03", 1, candidate, "s25")] = (
            "PARTIAL", [OMITTED], "KV 캐시 계층의 대상이 NVIDIA 추론 컨텍스트 메모리 스토리지 플랫폼이라는 선행 의미가 빠졌다."
        )
        judgments[key("N03", 1, candidate, "s61")] = (
            "FAIL", [CHANGED, EVIDENCE, MODALITY],
            "s62의 AI 정의 주행 도입 예정을 추가하면서 근거를 s61로 잘못 연결했고, 예측을 PLAN_OR_TARGET으로 분류했다.",
        )
    judgments[key("N03", 2, 1, "s25")] = (
        "FAIL", [EVIDENCE], "플랫폼 이름을 풀어 썼지만 이를 확인하는 s24를 연결하지 않았다."
    )
    judgments[key("N03", 2, 2, "s25")] = (
        "PARTIAL", [OMITTED], "KV 캐시 계층의 대상이 NVIDIA 추론 컨텍스트 메모리 스토리지 플랫폼이라는 선행 의미가 빠졌다."
    )
    for candidate in (1, 2):
        judgments[key("N03", 2, candidate, "s61")] = (
            "PARTIAL", [MODALITY], "조만간 볼 수 있다는 예측 문장은 보존했지만 modality를 PLAN_OR_TARGET으로 분류했다."
        )

    judgments[key("N04", 2, 1, "s4")] = (
        "PARTIAL", [MODALITY], "사우라브 쿨카니의 평가 발언은 보존했지만 modality를 FACT로 분류했다."
    )
    judgments[key("N04", 2, 1, "s25")] = (
        "PARTIAL", [OMITTED], "s24를 연결했지만 문장의 ‘해당 툴킷’을 Dell Omnia로 풀지 않았다."
    )
    judgments[key("N04", 2, 2, "s1")] = (
        "PARTIAL", [OMITTED], "발표일인 2025년 5월 21일을 생략했다."
    )
    judgments[key("N04", 2, 2, "s4")] = (
        "PARTIAL", [MODALITY], "사우라브 쿨카니의 평가 발언은 보존했지만 modality를 FACT로 분류했다."
    )

    for row in review["ratings"]:
        if row["verdict"] == "CONTRACT_ERROR":
            continue
        verdict, errors, reason = judgments.get(
            key(row["article_id"], row["repeat"], row["candidate"], row["source_id"]),
            ("PASS", [], "동결한 필수 의미·modality와 연결 근거를 보존했다."),
        )
        row.update(verdict=verdict, error_types=errors, reason=reason)
    if len(review["ratings"]) != 64 or any(row["verdict"] == "PENDING" for row in review["ratings"]):
        raise ValueError("REVIEW_INCOMPLETE")
    output = ROOT / "blind-review.json"
    encoded = json.dumps(review, ensure_ascii=False, indent=2) + "\n"
    if output.exists() and output.read_text(encoding="utf-8") != encoded:
        raise ValueError("BLIND_REVIEW_CHANGED")
    output.write_text(encoded, encoding="utf-8")
    counts = Counter(row["verdict"] for row in review["ratings"])
    print(json.dumps({"status": "BLIND_REVIEW_RECORDED", "ratings": len(review["ratings"]), "counts": counts}, ensure_ascii=False))


if __name__ == "__main__":
    main()
