"""Local diagnostic contracts; structural validity does not establish truth."""

from __future__ import annotations

import importlib.util
import sys
from pathlib import Path
from typing import Annotated, Literal

from pydantic import Field, model_validator


_NAME = "ontology_map_reassessment_diagnostics"
if _NAME not in sys.modules:
    _path = Path(__file__).resolve().parent.parent / "model-studio-reassessment" / "diagnostics.py"
    _spec = importlib.util.spec_from_file_location(_NAME, _path)
    _module = importlib.util.module_from_spec(_spec)
    sys.modules[_NAME] = _module
    _spec.loader.exec_module(_module)
_diagnostics = sys.modules[_NAME]
Source, Claim = _diagnostics.Source, _diagnostics.Claim
Strict, Text = _diagnostics.Strict, _diagnostics.Text


def _validate_sources_and_claims(article: list[Source], claims: list[Claim]) -> None:
    source_ids = _diagnostics._unique([source.source_id for source in article], "SOURCE_ID")
    _diagnostics._unique([claim.claim_id for claim in claims], "CLAIM_ID")
    for source in article:
        if source.end_char - source.start_char != len(source.text):
            raise ValueError("SOURCE_SPAN_LENGTH_MISMATCH")
    for claim in claims:
        _diagnostics._references(
            claim.main_source_ids + claim.context_source_ids, source_ids, "CLAIM_EVIDENCE"
        )


class EvidenceInput(Strict):
    article: list[Source]
    claim: Claim

    @model_validator(mode="after")
    def validate_references(self) -> EvidenceInput:
        _validate_sources_and_claims(self.article, [self.claim])
        return self


class AtomicInput(Strict):
    article: list[Source]
    claims: list[Claim]

    @model_validator(mode="after")
    def validate_references(self) -> AtomicInput:
        _validate_sources_and_claims(self.article, self.claims)
        return self


class EvidenceOutput(Strict):
    claim_id: Text
    verdict: Literal["PASS", "FAIL", "UNRESOLVED"]
    basis_source_ids: list[Text]
    reason: Text


class AtomicOutput(Strict):
    claims: list[EvidenceOutput]


class DecomposedClaim(EvidenceOutput):
    independent_statements: Annotated[list[Text], Field(min_length=1)]


class DecomposedOutput(Strict):
    claims: list[DecomposedClaim]


def validate_evidence(output: dict, inputs: dict) -> dict:
    context = EvidenceInput.model_validate(inputs)
    parsed = EvidenceOutput.model_validate(output)
    if parsed.claim_id != context.claim.claim_id:
        raise ValueError("WRONG_CLAIM_ID")
    allowed = set(context.claim.main_source_ids + context.claim.context_source_ids)
    _diagnostics._references(parsed.basis_source_ids, allowed, "EVIDENCE_BASIS")
    if parsed.verdict == "PASS" and not parsed.basis_source_ids:
        raise ValueError("PASS_WITHOUT_EVIDENCE")
    return parsed.model_dump()


def _validate_atomic_output(parsed: AtomicOutput | DecomposedOutput, context: AtomicInput) -> None:
    _diagnostics._exact(
        [claim.claim_id for claim in parsed.claims],
        {claim.claim_id for claim in context.claims},
        "CLAIM_REVIEW",
    )
    allowed = {source.source_id for source in context.article}
    for claim in parsed.claims:
        _diagnostics._references(claim.basis_source_ids, allowed, "ATOMIC_BASIS")


def validate_atomic(output: dict, inputs: dict) -> dict:
    context = AtomicInput.model_validate(inputs)
    parsed = AtomicOutput.model_validate(output)
    _validate_atomic_output(parsed, context)
    return parsed.model_dump()


def validate_decomposed(output: dict, inputs: dict) -> dict:
    context = AtomicInput.model_validate(inputs)
    parsed = DecomposedOutput.model_validate(output)
    _validate_atomic_output(parsed, context)
    for claim in parsed.claims:
        if claim.verdict == "PASS" and len(claim.independent_statements) != 1:
            raise ValueError("PASS_WITH_MULTIPLE_INDEPENDENT_STATEMENTS")
    return parsed.model_dump()
