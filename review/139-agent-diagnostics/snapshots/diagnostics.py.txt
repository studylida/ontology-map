"""Gold-free diagnostic contracts; validation checks structure, never truth."""

from __future__ import annotations

from typing import Annotated, Literal

from pydantic import BaseModel, ConfigDict, Field, StringConstraints


Text = Annotated[str, StringConstraints(min_length=1, pattern=r"\S")]
CHECK_KINDS = {"SOURCE_SUPPORT", "EVIDENCE_SUFFICIENCY", "ATOMICITY", "CITATION_SCOPE"}


class Strict(BaseModel):
    model_config = ConfigDict(extra="forbid", strict=True)


class Source(Strict):
    source_id: Text
    text: Text
    start_char: Annotated[int, Field(ge=0)]
    end_char: Annotated[int, Field(gt=0)]


class Claim(Strict):
    claim_id: Text
    statement: Text
    main_source_ids: list[Text]
    context_source_ids: list[Text]


class DiagnosticInput(Strict):
    article: Annotated[list[Source], Field(min_length=1)]
    focus_source_ids: Annotated[list[Text], Field(min_length=1)]
    claims: list[Claim]


class PreservedFact(Strict):
    statement: Text
    basis_source_ids: Annotated[list[Text], Field(min_length=1)]
    claim_ids: list[Text]
    verdict: Literal["PRESERVED", "PARTIAL", "MISSING", "DISTORTED", "UNRESOLVED"]
    reason: str


class SourcePreservation(Strict):
    source_id: Text
    decision: Literal["INCLUDE", "OMIT_ALLOWED", "UNRESOLVED"]
    reason: Text
    facts: list[PreservedFact]


class PreservationOutput(Strict):
    sources: list[SourcePreservation]


class FidelityCheck(Strict):
    kind: Literal["SOURCE_SUPPORT", "EVIDENCE_SUFFICIENCY", "ATOMICITY", "CITATION_SCOPE"]
    verdict: Literal["PASS", "FAIL", "UNRESOLVED"]
    basis_source_ids: list[Text]
    reason: str


class ClaimFidelity(Strict):
    claim_id: Text
    checks: list[FidelityCheck]


class FidelityOutput(Strict):
    claims: list[ClaimFidelity]


def _unique(values: list[str], name: str) -> set[str]:
    result = set(values)
    if len(result) != len(values):
        raise ValueError(f"DUPLICATE_{name}")
    return result


def _references(values: list[str], allowed: set[str], name: str) -> None:
    if not _unique(values, name) <= allowed:
        raise ValueError(f"UNKNOWN_{name}")


def _exact(values: list[str], expected: set[str], name: str) -> None:
    if _unique(values, name) != expected:
        raise ValueError(f"INCOMPLETE_{name}")


def validate_input(inputs: dict) -> DiagnosticInput:
    """Accept only source text, source focus and one actual candidate set."""
    parsed = DiagnosticInput.model_validate(inputs)
    source_ids = _unique([item.source_id for item in parsed.article], "SOURCE_ID")
    _unique([item.claim_id for item in parsed.claims], "CLAIM_ID")
    _references(parsed.focus_source_ids, source_ids, "FOCUS_SOURCE_ID")
    for source in parsed.article:
        if source.end_char - source.start_char != len(source.text):
            raise ValueError("SOURCE_SPAN_LENGTH_MISMATCH")
    for claim in parsed.claims:
        _references(claim.main_source_ids, source_ids, "MAIN_SOURCE_ID")
        _references(claim.context_source_ids, source_ids, "CONTEXT_SOURCE_ID")
        if set(claim.main_source_ids) & set(claim.context_source_ids):
            raise ValueError("DUPLICATE_EVIDENCE_ROLE")
    return parsed


def _validate_fact(fact: PreservedFact, source_ids: set[str], claim_ids: set[str]) -> None:
    _references(fact.basis_source_ids, source_ids, "FACT_BASIS")
    _references(fact.claim_ids, claim_ids, "FACT_CLAIM")
    if fact.verdict == "MISSING" and fact.claim_ids:
        raise ValueError("MISSING_WITH_CLAIM_MAPPING")
    if fact.verdict in {"PRESERVED", "PARTIAL", "DISTORTED"} and not fact.claim_ids:
        raise ValueError("JUDGED_FACT_WITHOUT_CLAIM_MAPPING")
    if fact.verdict != "PRESERVED" and not fact.reason.strip():
        raise ValueError("NONPASS_WITHOUT_REASON")


def validate_preservation(output: dict, inputs: dict) -> dict:
    """Return normalized structural output; semantic verdicts remain unverified."""
    context = validate_input(inputs)
    parsed = PreservationOutput.model_validate(output)
    _exact([item.source_id for item in parsed.sources], set(context.focus_source_ids), "FOCUS_REVIEW")
    source_ids = {item.source_id for item in context.article}
    claim_ids = {item.claim_id for item in context.claims}
    for source in parsed.sources:
        if (source.decision == "INCLUDE") != bool(source.facts):
            raise ValueError("DECISION_FACTS_MISMATCH")
        for fact in source.facts:
            _validate_fact(fact, source_ids, claim_ids)
    return parsed.model_dump()


def validate_fidelity(output: dict, inputs: dict) -> dict:
    """Check coverage and references without treating a model PASS as truth."""
    context = validate_input(inputs)
    parsed = FidelityOutput.model_validate(output)
    _exact([item.claim_id for item in parsed.claims], {item.claim_id for item in context.claims}, "CLAIM_REVIEW")
    source_ids = {item.source_id for item in context.article}
    for claim in parsed.claims:
        _exact([item.kind for item in claim.checks], CHECK_KINDS, "CHECK_KIND")
        for check in claim.checks:
            _references(check.basis_source_ids, source_ids, "CHECK_BASIS")
            if check.verdict != "PASS" and (not check.basis_source_ids or not check.reason.strip()):
                raise ValueError("NONPASS_WITHOUT_BASIS_OR_REASON")
    return parsed.model_dump()
