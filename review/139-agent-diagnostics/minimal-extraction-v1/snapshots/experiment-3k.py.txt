def read(path: Path) -> dict | list:
    return json.loads(path.read_text(encoding="utf-8"))


def digest(path: Path) -> str:
    return hashlib.sha256(path.read_bytes()).hexdigest()


def save_once(path: Path, value: object) -> None:
    encoded = json.dumps(value, ensure_ascii=False, indent=2) + "\n"
    if path.exists() and path.read_text(encoding="utf-8") != encoded:
        raise ValueError(f"FROZEN_FILE_CHANGED:{path.name}")
    if not path.exists():
        path.write_text(encoded, encoding="utf-8")


def validate_source_doc(doc: dict) -> None:
    if hashlib.sha256(doc["source"].encode()).hexdigest() != doc["sha256"]:
        raise ValueError("SOURCE_HASH_CHANGED")
    if not 1_200 <= len(doc["source"]) <= 20_000 or "AI 요약" in doc["source"]:
        raise ValueError("SOURCE_BODY_INVALID")
    unit_ids = [unit["block_id"] for unit in doc["sentences"]]
    if len(unit_ids) != len(set(unit_ids)):
        raise ValueError("DUPLICATE_SOURCE_UNIT")
    if any(doc["source"][unit["start_char"]:unit["end_char"]] != unit["text"] for unit in doc["sentences"]):
        raise ValueError("SOURCE_SLICE_CHANGED")


def source_docs() -> dict[str, dict]:
    docs = read(SOURCES)
    if not isinstance(docs, list) or len(docs) != 4:
        raise ValueError("SOURCE_SET_CHANGED")
    result = {doc["id"]: doc for doc in docs}
    if len(result) != 4:
        raise ValueError("DUPLICATE_SOURCE_ID")
    for doc in docs:
        validate_source_doc(doc)
    return result


def validate_gold_fact(fact: dict, lookup: set[str]) -> None:
    required = (fact["required_semantics"], fact["accepted_modalities"], fact["sufficient_evidence_sets"], fact["atomicity"])
    if not all(required):
        raise ValueError("INVALID_GOLD_FACT")
    if any(not set(group) <= lookup for group in fact["sufficient_evidence_sets"]):
        raise ValueError("UNKNOWN_GOLD_EVIDENCE")


def validate_gold_article(article: dict, docs: dict[str, dict]) -> list[str]:
    facts = article["facts"]
    if not 5 <= len(facts) <= 6:
        raise ValueError("GOLD_FACT_COUNT_CHANGED")
    lookup = {unit["block_id"] for unit in docs[article["article_id"]]["sentences"]}
    for fact in facts:
        validate_gold_fact(fact, lookup)
    return [fact["fact_id"] for fact in facts]


def gold_articles() -> dict[str, dict]:
    docs = source_docs()
    value = read(GOLD)
    articles = value["articles"]
    if len(articles) != 4 or {item["article_id"] for item in articles} != set(docs):
        raise ValueError("GOLD_ARTICLE_SET_CHANGED")
    fact_ids = [fact_id for article in articles for fact_id in validate_gold_article(article, docs)]
    if len(fact_ids) != 21 or len(fact_ids) != len(set(fact_ids)):
        raise ValueError("GOLD_FACT_SET_CHANGED")
    return {item["article_id"]: item for item in articles}


def article_input(doc: dict) -> dict:
    return {"article": [{
        "source_id": unit["block_id"],
        "start_char": unit["start_char"],
        "end_char": unit["end_char"],
        "text": unit["text"],
    } for unit in doc["sentences"]]}


def prompts() -> dict[str, str]:
    preserve = PRESERVE.read_text(encoding="utf-8").strip()
    return {
        "rich": BASE_PROMPT + "\n\n" + preserve + "\n\n" + RICH_RULE,
        "minimal": BASE_PROMPT + "\n\n" + preserve + "\n\n" + MINIMAL_RULE,
    }


def evidence(source_ids: list[str], lookup: dict[str, dict]) -> list[dict]:
    if not source_ids or len(source_ids) != len(set(source_ids)) or not set(source_ids) <= set(lookup):
        raise ValueError("INVALID_SOURCE_IDS")
    order = {source_id: index for index, source_id in enumerate(lookup)}
    return [{
        "source_id": source_id,
        "start_char": lookup[source_id]["start_char"],
        "end_char": lookup[source_id]["end_char"],
        "quote_text": lookup[source_id]["text"],
        "quote_hash": hashlib.sha256(lookup[source_id]["text"].encode()).hexdigest(),
    } for source_id in sorted(source_ids, key=order.__getitem__)]


def validate_unresolved(groups: list[dict], lookup: dict[str, dict]) -> list[list[str]]:
    result = []
    for group in groups:
        ids = group["source_ids"]
        if not ids or len(ids) != len(set(ids)) or not set(ids) <= set(lookup):
            raise ValueError("INVALID_UNRESOLVED_IDS")
        result.append(ids)
    return result


def normalized_minimal_claim(number: int, claim: dict, lookup: dict[str, dict]) -> dict:
    if not claim["statement"].strip():
        raise ValueError("EMPTY_STATEMENT")
    return {
        "claim_no": number,
        "statement": claim["statement"],
        "modality": claim["modality"],
        "source_ids": [item["source_id"] for item in evidence(claim["source_ids"], lookup)],
    }


def validate_minimal(output: dict, inputs: dict) -> dict:
    lookup = {unit["source_id"]: unit for unit in inputs["article"]}
    claims = [normalized_minimal_claim(number, claim, lookup) for number, claim in enumerate(output["claims"], 1)]
    return {"claims": claims, "unresolved": validate_unresolved(output["unresolved"], lookup)}


def validate_rich_element(element: dict, all_ids: list[str]) -> None:
    ids = element["source_ids"]
    if not element["value"].strip() or not ids or len(ids) != len(set(ids)) or not set(ids) <= set(all_ids):
        raise ValueError("INVALID_CRITICAL_ELEMENT")


def normalized_rich_claim(number: int, claim: dict, lookup: dict[str, dict]) -> dict:
    main_ids, context_ids = claim["main_source_ids"], claim["context_source_ids"]
    if (not claim["statement"].strip() or not claim["critical_elements"] or not main_ids
            or set(main_ids) & set(context_ids)):
        raise ValueError("INVALID_RICH_CLAIM")
    all_ids = main_ids + context_ids
    refs = evidence(all_ids, lookup)
    for element in claim["critical_elements"]:
        validate_rich_element(element, all_ids)
    return {
        "claim_no": number,
        "statement": claim["statement"],
        "modality": claim["modality"],
        "source_ids": [item["source_id"] for item in refs],
    }


def validate_rich(output: dict, inputs: dict) -> dict:
    lookup = {unit["source_id"]: unit for unit in inputs["article"]}
    claims = [normalized_rich_claim(number, claim, lookup) for number, claim in enumerate(output["claims"], 1)]
    return {"claims": claims, "unresolved": validate_unresolved(output["unresolved"], lookup)}


def jobs() -> list[dict]:
    docs, prompt_by_arm = source_docs(), prompts()
    shapes = {"rich": (RichOutput, validate_rich), "minimal": (MinimalOutput, validate_minimal)}
    result = []
    for index, doc in enumerate(docs.values()):
        for repeat in REPEATS:
            arms = ("rich", "minimal") if (index + repeat) % 2 else ("minimal", "rich")
            for arm in arms:
                shape, validator = shapes[arm]
                result.append({
                    "identity": f"minimal-extraction-v1-{doc['id']}-r{repeat}-{arm}",
                    "model": MODEL,
                    "shape": shape,
                    "prompt": prompt_by_arm[arm],
                    "inputs": article_input(doc),
                    "maximum": MAX_TOKENS,
                    "thinking": False,
                    "validator": validator,
                    "category": "heldout",
                })
    if len(result) != 16 or len({job["identity"] for job in result}) != 16:
        raise ValueError("JOB_SET_CHANGED")
    return result


def strict_payload(model: str, shape: type, prompt: str, inputs: dict, maximum: int, thinking: bool) -> dict:
    if thinking or maximum != MAX_TOKENS or model != MODEL:
        raise ValueError("MODEL_SETTINGS_CHANGED")
    arm = "minimal" if shape is MinimalOutput else "rich"
    return {
        "model": model,
        "messages": [{"role": "system", "content": prompt}, {"role": "user", "content": json.dumps(inputs, ensure_ascii=False)}],
        "response_format": {
            "type": "json_schema",
            "json_schema": {"name": f"knowledge_extraction_{arm}_v1", "strict": True, "schema": shape.model_json_schema()},
        },
        "enable_thinking": False,
        "stream": False,
        "max_tokens": maximum,
        "temperature": 0,
    }


def request_record(job: dict) -> dict:
    payload = strict_payload(job["model"], job["shape"], job["prompt"], job["inputs"], job["maximum"], job["thinking"])
    data = json.dumps(payload, ensure_ascii=False).encode()
    return {
        "identity": job["identity"],
        "request_sha256": hashlib.sha256(data).hexdigest(),
        "reservation_usd": float(base.reservation(data, job["model"], job["maximum"])),
    }


def prior_overlap() -> dict:
    previous_docs = read(PREVIOUS / "sources-frozen.json") + read(ROOT.parent / "model-studio-preserving-extraction" / "sources-frozen.json") + read(LONG / "sources-frozen.json")
    hashes = {doc["sha256"] for doc in previous_docs}
    urls = {doc.get("url") or (doc.get("metadata") or {}).get("url") for doc in previous_docs}
    current = source_docs().values()
    return {
        "comparison_docs": len(previous_docs),
        "exact_hash_matches": sorted(doc["id"] for doc in current if doc["sha256"] in hashes),
        "exact_url_matches": sorted(doc["id"] for doc in current if doc["url"] in urls),
        "event_family_review": "Four distinct official events; manually checked against the previous twenty article families",
        "event_family_matches": [],
    }


def manifest() -> dict:
    planned = [request_record(job) for job in jobs()]
    reservation = sum((Decimal(str(item["reservation_usd"])) for item in planned), Decimal(0))
    overlap = prior_overlap()
    if any(overlap[key] for key in ("exact_hash_matches", "exact_url_matches", "event_family_matches")):
        raise ValueError("PREVIOUS_SOURCE_EXPOSURE")
    return {
        "version": 1,
        "issue": 139,
        "purpose": "Compare rich diagnostic and minimal runtime outputs during whole-article original-preserving extraction",
        "model": MODEL,
        "settings": {"temperature": 0, "enable_thinking": False, "stream": False, "max_tokens": MAX_TOKENS, "strict_json_schema": True},
        "arms": {
            "rich": "statement, modality, critical elements, main/context source IDs",
            "minimal": "statement, modality, undifferentiated source IDs",
        },
        "articles": 4,
        "pre_frozen_required_facts": 21,
        "repeats": list(REPEATS),
        "planned_calls": 16,
        "planned_reservation_usd": float(reservation),
        "historical_reserved_or_estimated_usd": float(HISTORICAL_USD),
        "cap_usd": float(CAP_USD),
        "maximum_cumulative_usd": float(HISTORICAL_USD + CAP_USD),
        "source_exposure_check": overlap,
        "success_criteria": {
            "comparable_articles_minimum": 3,
            "minimal_net_strict_fact_pass_minimum": 5,
            "new_minimal_material_regressions": 0,
            "unsupported_claim_rate_must_not_increase": True,
        },
        "dependency_hashes": {
            name: digest(path) for name, path in {
                "sources-frozen.json": SOURCES,
                "gold.json": GOLD,
                "preserve-addendum.txt": PRESERVE,
                "experiment.py": ROOT / "experiment.py",
                "trial.py": PIPELINE / "trial.py",
            }.items()
        },
        "requests": planned,
        "stop_rules": [
            "Never repair or retry a failed request",
            "Continue only after an isolated output or local validation contract error with known usage",
            "Stop on authentication, transport, provider shape, unknown cost, budget, frozen input, or source integrity failure",
            "Do not tune against these outputs and recount the articles as independent evaluation",
            "Do not count omission or unresolved output as PASS",
            "Do not infer a product contract, worker, automatic persistence, or operational quality",
        ],
    }


def verify_manifest() -> dict:
    frozen = read(MANIFEST)
    if frozen != manifest():
        raise ValueError("MANIFEST_CHANGED")
    return frozen


def configure_ledger() -> None:
    base.ROOT = ROOT
    base.HISTORICAL_USD = HISTORICAL_USD
    base.DEVELOPMENT_CAP_USD = Decimal("0")
    base.HELDOUT_CAP_USD = CAP_USD
    if not (ROOT / "ledger.json").exists():
        save_once(ROOT / "ledger.json", base.initial_ledger())


def prepare() -> None:
    gold_articles()
    frozen = manifest()
    if Decimal(str(frozen["planned_reservation_usd"])) > CAP_USD or frozen["maximum_cumulative_usd"] >= 1:
        raise ValueError("BUDGET_STOP")
    save_once(MANIFEST, frozen)
    configure_ledger()
    print(json.dumps({
        "status": "OFFLINE_READY",
        "manifest_sha256": digest(MANIFEST),
        "planned_calls": frozen["planned_calls"],
        "planned_reservation_usd": frozen["planned_reservation_usd"],
        "maximum_cumulative_usd": frozen["maximum_cumulative_usd"],
        "source_exposure_check": frozen["source_exposure_check"],
    }, ensure_ascii=False, indent=2))


def continuable_contract_failure(row: dict) -> bool:
    if row.get("status") != "CONTRACT_ERROR":
        return False
    if row.get("error_type") == "ValidationError":
        return True
    return row.get("error_type") == "ValueError" and row.get("reason") not in {
        "USAGE_UNKNOWN", "USAGE_BOUND_EXCEEDED", "RESPONSE_MODEL_MISMATCH", "UNEXPECTED_THINKING",
    }


def run_job(job: dict) -> None:
    expected = request_record(job)
    previous = {row["identity"]: row for row in base.results()}.get(job["identity"])
    if previous:
        if previous.get("request_sha256") != expected["request_sha256"]:
            raise ValueError(f"PREVIOUS_REQUEST_CHANGED:{job['identity']}")
        return
    try:
        base.call(**job)
    except ValueError as error:
        row = {item["identity"]: item for item in base.results()}[job["identity"]]
        if str(error) != "CALL_REQUIRES_REVIEW" or not continuable_contract_failure(row):
            raise


def run(comment_id: int) -> None:
    if comment_id <= 0:
        raise ValueError("ISSUE_COMMENT_REQUIRED")
    with (ROOT / "paid-run.lock").open("w") as lock:
        fcntl.flock(lock, fcntl.LOCK_EX | fcntl.LOCK_NB)
        verify_manifest()
        save_once(AUTHORIZATION, {"issue": 139, "comment_id": comment_id, "manifest_sha256": digest(MANIFEST)})
        configure_ledger()
        base.reconcile_pending()
        base.credentials.load_key()
        original_payload = base.build_payload
        base.build_payload = strict_payload
        try:
            for job in jobs():
                run_job(job)
        finally:
            base.build_payload = original_payload
            os.environ.pop("DASHSCOPE_API_KEY", None)


def public_candidate(row: dict) -> dict:
    if row["status"] == "PASS":
        return {"status": "PASS", **row["normalized"]}
    return {"status": row["status"], "error_type": row.get("error_type"), "finish_reason": row.get("finish_reason")}


def blind_arms(article_id: str, repeat: int) -> list[str]:
    arms = ["rich", "minimal"]
    if int(hashlib.sha256(f"{article_id}:{repeat}:minimal-v1".encode()).hexdigest(), 16) % 2:
        arms.reverse()
    return arms


def build_attempts(docs: dict[str, dict], rows: dict[str, dict]) -> tuple[list[dict], dict[str, list[str]]]:
    attempts, mapping = [], {}
    for article_id in docs:
        for repeat in REPEATS:
            arms = blind_arms(article_id, repeat)
            mapping[f"{article_id}:r{repeat}"] = arms
            attempts.append({
                "article_id": article_id,
                "repeat": repeat,
                "candidate_1": public_candidate(rows[f"minimal-extraction-v1-{article_id}-r{repeat}-{arms[0]}"]),
                "candidate_2": public_candidate(rows[f"minimal-extraction-v1-{article_id}-r{repeat}-{arms[1]}"]),
            })
    return attempts, mapping


def review_template(attempts: list[dict], gold: dict[str, dict]) -> dict:
    fact_ratings, claim_ratings = [], []
    for attempt in attempts:
        for candidate_no in (1, 2):
            candidate = attempt[f"candidate_{candidate_no}"]
            failed = candidate["status"] != "PASS"
            fact_ratings.extend({
                "article_id": attempt["article_id"], "repeat": attempt["repeat"], "candidate": candidate_no,
                "fact_id": fact["fact_id"], "verdict": "CONTRACT_ERROR" if failed else "PENDING",
                "error_types": [], "mapped_claim_nos": [], "reason": "호출 단위 출력 계약 실패" if failed else "",
            } for fact in gold[attempt["article_id"]]["facts"])
            if not failed:
                claim_ratings.extend({
                    "article_id": attempt["article_id"], "repeat": attempt["repeat"], "candidate": candidate_no,
                    "claim_no": claim["claim_no"], "verdict": "PENDING", "error_types": [], "reason": "",
                } for claim in candidate["claims"])
    return {
        "reviewer": "Codex manual semantic review against pre-frozen gold",
        "conditions_revealed": False,
        "fact_ratings": fact_ratings,
        "claim_ratings": claim_ratings,
    }


def packet() -> None:
    verify_manifest()
    configure_ledger()
    rows = {row["identity"]: row for row in base.results()}
    if set(rows) != {job["identity"] for job in jobs()}:
        raise ValueError("RESULT_SET_INCOMPLETE")
    docs, gold = source_docs(), gold_articles()
    attempts, mapping = build_attempts(docs, rows)
    save_once(ROOT / "blind-packet.json", {
        "instructions": "동결 gold로 필수 사실과 생성된 Claim을 모두 평가한다. blind-map.json은 판정을 동결하기 전에 열지 않는다.",
        "sources": {article_id: article_input(doc) for article_id, doc in docs.items()},
        "gold": read(GOLD),
        "attempts": attempts,
    })
    save_once(ROOT / "blind-map.json", mapping)
    template = review_template(attempts, gold)
    save_once(ROOT / "review-template.json", template)
    print(json.dumps({"status": "BLIND_PACKET_READY", "attempts": len(attempts), "fact_ratings": len(template["fact_ratings"]), "claim_ratings": len(template["claim_ratings"])}, ensure_ascii=False))


def validate_fact_rating(rating: dict) -> None:
    if rating["verdict"] not in {"PASS", "PARTIAL", "FAIL", "MISSING", "UNRESOLVED", "CONTRACT_ERROR"}:
        raise ValueError("INVALID_FACT_VERDICT")
    if not set(rating["error_types"]) <= ERROR_TYPES or (rating["verdict"] == "PASS" and rating["error_types"]):
        raise ValueError("INVALID_FACT_ERRORS")
    if rating["verdict"] != "PASS" and not rating["reason"]:
        raise ValueError("NONPASS_WITHOUT_REASON")


def validate_claim_rating(rating: dict) -> None:
    if rating["verdict"] not in {"SUPPORTED", "UNSUPPORTED", "OVERMERGED", "UNRESOLVED"}:
        raise ValueError("INVALID_CLAIM_VERDICT")
    if rating["verdict"] != "SUPPORTED" and not rating["reason"]:
        raise ValueError("NONSUPPORTED_WITHOUT_REASON")


def validate_review(review: dict, packet_value: dict) -> None:
    if review["conditions_revealed"] or len(review["fact_ratings"]) != 84:
        raise ValueError("REVIEW_SET_CHANGED")
    expected_claims = sum(len(attempt[f"candidate_{number}"].get("claims", [])) for attempt in packet_value["attempts"] for number in (1, 2))
    if len(review["claim_ratings"]) != expected_claims:
        raise ValueError("CLAIM_REVIEW_SET_CHANGED")
    for rating in review["fact_ratings"]:
        validate_fact_rating(rating)
    for rating in review["claim_ratings"]:
        validate_claim_rating(rating)


def aggregate_facts(review: dict, mapping: dict) -> tuple[dict, dict, dict, dict]:
    totals: dict[str, Counter] = defaultdict(Counter)
    errors: dict[str, Counter] = defaultdict(Counter)
    attempts: dict[str, dict[tuple[str, int], list[str]]] = defaultdict(lambda: defaultdict(list))
    lookup = {}
    for rating in review["fact_ratings"]:
        arm = mapping[f"{rating['article_id']}:r{rating['repeat']}"][rating["candidate"] - 1]
        totals[arm][rating["verdict"]] += 1
        errors[arm].update(rating["error_types"])
        attempts[arm][(rating["article_id"], rating["repeat"])].append(rating["verdict"])
        lookup[(rating["article_id"], rating["repeat"], rating["fact_id"], arm)] = rating
    return totals, errors, attempts, lookup


def aggregate_claims(review: dict, mapping: dict) -> dict[str, Counter]:
    totals: dict[str, Counter] = defaultdict(Counter)
    for rating in review["claim_ratings"]:
        arm = mapping[f"{rating['article_id']}:r{rating['repeat']}"][rating["candidate"] - 1]
        totals[arm][rating["verdict"]] += 1
    return totals


def paired_facts(lookup: dict) -> tuple[Counter, list[dict]]:
    rank = {"FAIL": 0, "MISSING": 0, "UNRESOLVED": 0, "PARTIAL": 1, "PASS": 2}
    paired, regressions = Counter(), []
    for key in sorted({item[:3] for item in lookup}):
        rich, minimal = lookup[key + ("rich",)], lookup[key + ("minimal",)]
        if "CONTRACT_ERROR" in (rich["verdict"], minimal["verdict"]):
            paired["excluded_contract"] += 1
        elif rank[minimal["verdict"]] > rank[rich["verdict"]]:
            paired["minimal_better"] += 1
        elif rank[minimal["verdict"]] < rank[rich["verdict"]]:
            paired["rich_better"] += 1
        else:
            paired["tie"] += 1
        if ("CONTRACT_ERROR" not in (rich["verdict"], minimal["verdict"])
                and set(minimal["error_types"]) & MATERIAL_ERRORS
                and not set(rich["error_types"]) & MATERIAL_ERRORS):
            regressions.append({"article_id": key[0], "repeat": key[1], "fact_id": key[2], "errors": minimal["error_types"]})
    return paired, regressions


def comparable_article_ids(rows: list[dict]) -> list[str]:
    status = {row["identity"]: row["status"] for row in rows}
    return [article_id for article_id in source_docs() if all(
        status[f"minimal-extraction-v1-{article_id}-r{repeat}-{arm}"] == "PASS"
        for repeat in REPEATS for arm in ("rich", "minimal")
    )]


def unsupported_rates(claim_totals: dict[str, Counter]) -> dict[str, float]:
    return {
        arm: (claim_totals[arm]["UNSUPPORTED"] / sum(claim_totals[arm].values()) if claim_totals[arm] else 0)
        for arm in ("rich", "minimal")
    }


def decision_for(comparable: list[str], net_pass: int, regressions: list[dict], rates: dict[str, float]) -> str:
    if len(comparable) < 3:
        return "DEFERRED_INSUFFICIENT_COMPARABLE_ARTICLES"
    if net_pass >= 5 and not regressions and rates["minimal"] <= rates["rich"]:
        return "KEEP_MINIMAL_RUNTIME_CONTRACT_CANDIDATE"
    return "REJECT_MINIMAL_RUNTIME_CONTRACT_CANDIDATE"


def score() -> None:
    review, mapping, packet_value = read(ROOT / "blind-review.json"), read(ROOT / "blind-map.json"), read(ROOT / "blind-packet.json")
    validate_review(review, packet_value)
    rows = [json.loads(line) for line in (ROOT / "results.jsonl").read_text(encoding="utf-8").splitlines()]
    totals, errors, attempt_facts, lookup = aggregate_facts(review, mapping)
    claim_totals = aggregate_claims(review, mapping)
    paired, regressions = paired_facts(lookup)
    comparable_articles = comparable_article_ids(rows)
    net_pass = totals["minimal"]["PASS"] - totals["rich"]["PASS"]
    unsupported_rate = unsupported_rates(claim_totals)
    decision = decision_for(comparable_articles, net_pass, regressions, unsupported_rate)
    ledger = read(ROOT / "ledger.json")
    result = {
        "version": 1,
        "issue": 139,
        "method": "Single-reviewer semantic review against pre-frozen gold with normalized candidates and arm names hidden until ratings were frozen",
        "facts": {arm: {"verdicts": dict(totals[arm]), "errors": dict(errors[arm]), "complete_attempts": sum(all(value == "PASS" for value in values) for values in attempt_facts[arm].values())} for arm in ("rich", "minimal")},
        "claims": {arm: {"verdicts": dict(claim_totals[arm]), "unsupported_rate": round(unsupported_rate[arm], 4)} for arm in ("rich", "minimal")},
        "paired_noncontract": dict(paired),
        "minimal_net_strict_fact_pass": net_pass,
        "new_minimal_material_regressions": regressions,
        "comparable_articles": comparable_articles,
        "decision": decision,
        "contract_failures": [{"identity": row["identity"], "error_type": row.get("error_type"), "finish_reason": row.get("finish_reason"), "reason": row.get("reason")} for row in rows if row["status"] != "PASS"],
        "cost": {
            "experiment_usd": ledger["heldout_reserved_or_actual_usd"],
            "historical_reserved_or_estimated_usd": ledger["historical_reserved_or_estimated_usd"],
            "cumulative_usd": round(ledger["historical_reserved_or_estimated_usd"] + ledger["heldout_reserved_or_actual_usd"], 8),
        },
        "interpretation_limit": "This purpose-sampled diagnostic does not approve a product output contract, worker, persistence, or operational quality.",
    }
    save_once(ROOT / "score.json", result)
    print(json.dumps({"status": "SCORED", "decision": decision, "net_pass": net_pass, "paired": dict(paired)}, ensure_ascii=False))


def main() -> None:
    parser = argparse.ArgumentParser()
    parser.add_argument("command", choices=("prepare", "run", "packet", "score"))
    parser.add_argument("--paid", action="store_true")
    parser.add_argument("--comment-id", type=int)
    args = parser.parse_args()
    if args.command == "prepare":
        prepare()
    elif args.command == "run":
        if not args.paid or args.comment_id is None:
            raise ValueError("EXPLICIT_PAID_FLAG_AND_COMMENT_REQUIRED")
        run(args.comment_id)
    elif args.command == "packet":
        packet()
    else:
        score()


if __name__ == "__main__":
    try:
        main()
    except Exception as error:  # noqa: BLE001 - suppress credential-bearing tracebacks
        print(json.dumps({"status": "STOPPED", "error_type": type(error).__name__, "reason": str(error) if isinstance(error, ValueError) else "details suppressed"}, ensure_ascii=False))
        raise SystemExit(1) from None
    finally:
        os.environ.pop("DASHSCOPE_API_KEY", None)
